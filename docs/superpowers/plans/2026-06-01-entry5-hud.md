Add before `Player::draw()` (line ~298), a new method:

```cpp
void Player::drawHealthBar()
{
    auto& atlas = BlockDatabase::get().textureAtlas;
    GLuint atlasID = atlas.getID();

    // Heart tile coordinates on atlas: full=(13,1), half=(14,1), empty=(15,1)
    auto fullUV  = atlas.getTexture(sf::Vector2i(13, 1));
    auto halfUV  = atlas.getTexture(sf::Vector2i(14, 1));
    auto emptyUV = atlas.getTexture(sf::Vector2i(15, 1));

    const float heartSize = 24.0f;
    const float gap = 2.0f;
    const float startX = 16.0f;
    const float startY = 16.0f;

    auto* fg = ImGui::GetForegroundDrawList();

    for (int i = 0; i < 10; i++) {
        float x = startX + i * (heartSize + gap);
        int hpForHeart = m_hp - i * 10; // each heart = 10 HP

        std::array<GLfloat, 8>* uv;
        if (hpForHeart >= 10)
            uv = &fullUV;
        else if (hpForHeart >= 5)
            uv = &halfUV;
        else
            uv = &emptyUV;

        ImVec2 uv0((*uv)[2], (*uv)[5]); // bottom-left in atlas UV
        ImVec2 uv1((*uv)[0], (*uv)[3]); // top-right in atlas UV
        ImVec2 p0(x, startY);
        ImVec2 p1(x + heartSize, startY + heartSize);

        fg->AddImage((ImTextureID)(intptr_t)atlasID, p0, p1, uv0, uv1);
    }
}
```

Note: The atlas `getTexture()` returns UVs as `{xMax, yMax, xMin, yMax, xMin, yMin, xMax, yMin}`. Index [2]=xMin, [5]=yMin (bottom-left), [0]=xMax, [3]=yMax (top-right). This matches the pattern used in `drawSlot` and drop item rendering.

```cpp
void Player::drawTimer()
{
    if (!m_roundActive) return;

    auto displaySize = ImGui::GetIO().DisplaySize;

    int minutes = (int)m_roundTimeLeft / 60;
    int seconds = (int)m_roundTimeLeft % 60;

    // Prepare Chinese round text via BitmapText (only when round number changes)
    static int lastRenderedRound = -1;
    static GLuint roundTexId = 0;
    static float roundTexW = 0, roundTexH = 0;
    if ((int)m_roundNumber != lastRenderedRound) {
        lastRenderedRound = (int)m_roundNumber;
        char buf[32];
        snprintf(buf, sizeof(buf), "第 %d 回合", (int)m_roundNumber);
        std::vector<std::string> lines = { buf };
        roundTexW = 120;
        roundTexH = 40;
        roundTexId = m_hudText.update(lines, (int)roundTexW, (int)roundTexH);
    }

    const float winW = 220.0f;
    const float winH = 36.0f;
    float winX = (displaySize.x - winW) * 0.5f;
    float winY = 8.0f;

    ImGui::SetNextWindowPos(ImVec2(winX, winY), ImGuiCond_Always);
    ImGui::SetNextWindowSize(ImVec2(winW, winH), ImGuiCond_Always);
    ImGui::PushStyleVar(ImGuiStyleVar_WindowPadding, ImVec2(0, 0));
    ImGui::PushStyleColor(ImGuiCol_WindowBg, ImVec4(0.0f, 0.0f, 0.0f, 0.4f));

    int flags = ImGuiWindowFlags_NoMove | ImGuiWindowFlags_NoResize |
                ImGuiWindowFlags_NoCollapse | ImGuiWindowFlags_NoTitleBar |
                ImGuiWindowFlags_NoBringToFrontOnFocus;

    if (ImGui::Begin("Timer", nullptr, flags)) {
        ImVec2 winPos = ImGui::GetWindowPos();
        auto* dl = ImGui::GetWindowDrawList();

        // Left: round text via BitmapText texture
        if (roundTexId != 0) {
            float textH = winH - 8.0f;
            float textW = textH * (roundTexW / roundTexH);
            dl->AddImage((ImTextureID)(intptr_t)roundTexId,
                         ImVec2(winPos.x + 8, winPos.y + 4),
                         ImVec2(winPos.x + 8 + textW, winPos.y + 4 + textH));
        }

        // Right: MM:SS timer
        char timeBuf[16];
        snprintf(timeBuf, sizeof(timeBuf), "%02d:%02d", minutes, seconds);
        bool redFlash = (m_roundTimeLeft <= 30.0f) && (sin(ImGui::GetTime() * 4.0f) > 0.0);
        ImU32 timerColor = redFlash ? IM_COL32(255, 60, 60, 255) : IM_COL32(255, 255, 255, 255);
        // Draw MM:SS right-aligned
        dl->AddText(ImGui::GetFont(), 20.0f,
                    ImVec2(winPos.x + winW - 56.0f, winPos.y + 6.0f),
                    timerColor, timeBuf);
    }
    ImGui::End();

    ImGui::PopStyleColor();
    ImGui::PopStyleVar();
}
```

Note: `static int lastRenderedRound` caches the round number and only re-renders BitmapText when it changes, avoiding TTF rasterization every frame. The `MM:SS` text is ASCII and renders with ImGui's default font at 20px. `sin(ImGui::GetTime() * 4)` gives ~2.5 Hz flash for the last 30 seconds.

```cpp
void Player::drawSettlement(const Camera* camera)
{
    // Trigger: round ended OR player dead
    bool shouldShow = (!m_roundActive && m_roundNumber > 0) || m_isDead;
    if (!shouldShow) return;

    auto displaySize = ImGui::GetIO().DisplaySize;

    // --- Prepare BitmapText lines ---
    char titleBuf[64];
    snprintf(titleBuf, sizeof(titleBuf), "═══ 第 %d 回合 结束 ═══", m_roundNumber);

    int survMin = (int)(600.0f - m_roundTimeLeft) / 60;
    int survSec = (int)(600.0f - m_roundTimeLeft) % 60;
    char survBuf[64];
    snprintf(survBuf, sizeof(survBuf), "存活时间   %02d:%02d", survMin, survSec);

    char killBuf[64];
    snprintf(killBuf, sizeof(killBuf), "击杀猪人   %d", m_pigmanKills);

    int totalItems = 0;
    int totalTypes = 0;
    for (auto& [matId, count] : m_roundCollection) {
        if (count > 0) { totalItems += count; totalTypes++; }
    }
    char collectBuf[64];
    snprintf(collectBuf, sizeof(collectBuf), "收集物品   %d 类 / %d 个", totalTypes, totalItems);

    // Build detail lines
    auto cnName = [](Material::ID id) -> std::string {
        switch (id) {
            case Material::ID::Nothing:    return "空";
            case Material::ID::Grass:      return "草方块";
            case Material::ID::Dirt:       return "泥土";
            case Material::ID::Stone:      return "石材";
            case Material::ID::OakBark:    return "木材";
            case Material::ID::OakLeaf:    return "树叶";
            case Material::ID::Sand:       return "沙子";
            case Material::ID::Cactus:     return "仙人掌";
            case Material::ID::Rose:       return "玫瑰";
            case Material::ID::TallGrass:  return "草";
            case Material::ID::DeadShrub:  return "枯木";
            case Material::ID::Stick:      return "木棍";
            case Material::ID::WoodenSword:return "木剑";
            case Material::ID::RawMeat:    return "生肉";
            default: return "未知";
        }
    };

    // Build collection detail lines (right-aligned count with padding)
    std::vector<std::string> detailLines;
    for (auto& [matId, count] : m_roundCollection) {
        if (count > 0) {
            char buf[64];
            snprintf(buf, sizeof(buf), "%s ×%d", cnName(matId).c_str(), count);
            detailLines.push_back(buf);
        }
    }
    int detailCount = (int)detailLines.size();

    // Batch all text lines into one BitmapText texture
    std::vector<std::string> textLines;
    textLines.push_back(titleBuf);
    textLines.push_back("");
    textLines.push_back(survBuf);
    textLines.push_back(killBuf);
    textLines.push_back(collectBuf);
    if (detailCount > 0) {
        textLines.push_back("");
        textLines.push_back("收集明细:");
        for (auto& dl : detailLines)
            textLines.push_back("  " + dl);
    }

    int texW = 320;
    int lineH = 28;
    int texH = 8 + (int)textLines.size() * lineH + 8;
    GLuint hudTexId = m_hudText.update(textLines, texW, texH);

    // --- Render full-screen overlay ---
    ImGui::SetNextWindowPos(ImVec2(0, 0), ImGuiCond_Always);
    ImGui::SetNextWindowSize(displaySize, ImGuiCond_Always);
    ImGui::PushStyleVar(ImGuiStyleVar_WindowPadding, ImVec2(0, 0));
    ImGui::PushStyleColor(ImGuiCol_WindowBg, ImVec4(0.0f, 0.0f, 0.0f, 0.7f));

    int flags = ImGuiWindowFlags_NoMove | ImGuiWindowFlags_NoResize |
                ImGuiWindowFlags_NoCollapse | ImGuiWindowFlags_NoTitleBar |
                ImGuiWindowFlags_NoBringToFrontOnFocus;

    if (ImGui::Begin("Settlement", nullptr, flags)) {
        // Center the panel
        float panelW = texW + 40.0f;
        float panelH = texH + 60.0f;
        float panelX = (displaySize.x - panelW) * 0.5f;
        float panelY = (displaySize.y - panelH) * 0.5f;

        auto* dl = ImGui::GetWindowDrawList();
        dl->AddRectFilled(ImVec2(panelX, panelY),
                          ImVec2(panelX + panelW, panelY + panelH),
                          IM_COL32(20, 20, 20, 220));
        dl->AddRect(ImVec2(panelX, panelY),
                    ImVec2(panelX + panelW, panelY + panelH),
                    IM_COL32(180, 180, 180, 255), 0.0f, 0, 2.0f);

        // Draw BitmapText texture (one big image)
        dl->AddImage((ImTextureID)(intptr_t)hudTexId,
                     ImVec2(panelX + 20, panelY + 20),
                     ImVec2(panelX + 20 + texW, panelY + 20 + texH));

        // Placeholder button at bottom
        float btnW = 160.0f, btnH = 36.0f;

        float btnY = panelY + panelH - btnH - 16.0f;
        dl->AddRectFilled(ImVec2(btnX, btnY),
                          ImVec2(btnX + btnW, btnY + btnH),

                    IM_COL32(180, 180, 180, 255),
```
